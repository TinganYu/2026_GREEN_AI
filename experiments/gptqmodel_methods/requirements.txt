# Installation order matters:
#   1. pip install torch --index-url https://download.pytorch.org/whl/cu128
#   2. pip install -v gptqmodel --no-build-isolation --force-reinstall --no-cache-dir
#   3. pip install -r requirements.txt
#
# Step 2 MUST use --no-build-isolation so GPTQModel compiles CUDA kernels
# (AWQ V2, QQQ) against the installed torch. Without it, only a pre-built
# wheel is installed and kernels will be missing.
